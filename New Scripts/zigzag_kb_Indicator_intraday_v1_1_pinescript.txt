//@version=6
indicator("ZZ KB Nested Swings INTRADAY V1.1", "ZZ KB INTRA1.1", overlay=true, max_lines_count=500, max_labels_count=500, max_boxes_count=500)
// WHAT'S NEW vs INTRADAY V1 (mirrors Nested Swings V2.1, 2026-07-06):
//   A. RETIRE MISSED TRADES — a WAITING trade whose whole target zone price closed
//      above is marked 4 "✖ missed" (permanent); the spotlight moves on.
//   B. Completed-trade labels toggle (dim "✓ Tn" target-hit / "✖ Tn" missed).
//   C. SEED-PIVOT FIX — the ZigZag library's degenerate first pivot no longer hides
//      setups whose extreme IS the chart's first pivot (matters MORE intraday:
//      limited 5m/15m history makes the first pivot recent).

// ============================================================================
// ZZ KB Nested Swings INTRADAY V1
// ============================================================================
//
// Same engine and trade rules as "ZZ KB Nested Swings V2" (the validated
// daily/long-only version), re-tuned for INTRADAY INDEX charts (5m / 15m).
// The strategy's shape is unchanged - only the yardstick shrinks:
//   trade gain per swing ~ 0.236 x decline size, so
//   required decline ~ desired gain / 0.236.
//
// WHAT'S DIFFERENT vs V2
//   1. Deviation inputs allow sub-1% values (floors lowered to 0.1%).
//   2. Intraday defaults: Min Decline 1.5%, Peak Sensitivity 1.0%,
//      Min Gap 0.35% (= Nifty/Sensex 15-minute starting point).
//   3. VOLUME AUTO-BYPASS: index symbols (NIFTY / BANKNIFTY / SENSEX) have no
//      volume data, which would silently block every entry. The filter now
//      detects this and bypasses itself; the info table shows "n/a (index)".
//      On futures/stocks (real volume) the filter works exactly as in V2.
//
// SUGGESTED STARTING VALUES (tune from observation, like the daily version)
//   Nifty / Sensex 15m : Decline 1.5   Peak 1.0   Gap 0.35
//   Nifty / Sensex  5m : Decline 1.0   Peak 0.7   Gap 0.25
//   Bank Nifty     15m : Decline 2.25  Peak 1.6   Gap 0.5
//   Bank Nifty      5m : Decline 1.5   Peak 1.0   Gap 0.35
//   (ratios preserved from V2: Peak = 0.71 x Decline, Gap = 0.23 x Decline)
//
// TRADE MATH (UNCHANGED - the validated rules)
//   Entry : confirmed close > 0.382  +  close > EMA9 & EMA21  +  volume filter
//   TP    : high reaches the golden zone 0.618-0.68 (first touch of 0.618)
//   SL    : confirmed close < 0.236     (all measured from B up to each peak A)
//
// SCOPE (user decision): the decline (setup) may build over 1-3 sessions;
// the entry->target leg is what you trade intraday.
// ============================================================================

import TradingView/ZigZag/7 as zigzag

// =============== INPUTS ===============

grpDetection = "Detection"
minDeclinePct = input.float(1.5, "Min Decline % (sets B / setup)", minval=0.1, maxval=80.0, step=0.1, group=grpDetection,
     tooltip="How deep a fall must be to count as a tradeable setup. Starting points - Nifty/Sensex: 1.5 (15m) / 1.0 (5m). Bank Nifty: 2.25 (15m) / 1.5 (5m). Rule of thumb: required decline = desired trade gain / 0.236.")
pivotSensPct = input.float(1.0, "Peak Sensitivity % (finds nested peaks)", minval=0.1, maxval=50.0, step=0.1, group=grpDetection,
     tooltip="Finer setting used ONLY to find the nested peaks. Keep at roughly 0.7x of Min Decline %. Nifty/Sensex: 1.0 (15m) / 0.7 (5m). Bank Nifty: 1.6 (15m) / 1.0 (5m).")
zigDepth = input.int(10, "Min Bars Between Pivots", minval=2, maxval=50, group=grpDetection,
     tooltip="On 15m, 10 bars = 2.5 hours. On 5m consider 12-15 to skip opening-burst noise.")
maxSwings = input.int(4, "Max Nested Swings", minval=1, maxval=4, group=grpDetection)
minGapPct = input.float(0.35, "Min Gap Between Peaks %", minval=0.0, maxval=50.0, step=0.05, group=grpDetection,
     tooltip="Each higher peak must be at least this % above the previous one; if two are closer, the HIGHER is kept. Keep at roughly 0.23x of Min Decline %.")

grpFib = "Fibonacci Levels"
fib_sl1     = input.float(0.236, "SL line",           minval=0.05, maxval=0.4,  step=0.001, group=grpFib)
fib_entry1L = input.float(0.32,  "Entry Zone — Low",  minval=0.1,  maxval=0.5,  step=0.001, group=grpFib)
fib_entry1H = input.float(0.382, "Entry Zone — High", minval=0.1,  maxval=0.5,  step=0.001, group=grpFib)
fib_tp1L    = input.float(0.618, "Golden TP — Low",   minval=0.5,  maxval=0.8,  step=0.001, group=grpFib)
fib_tp1H    = input.float(0.68,  "Golden TP — High",  minval=0.5,  maxval=0.95, step=0.001, group=grpFib)

grpEMA = "Moving Averages"
showEMAs    = input.bool(false,       "Show EMA 9 / EMA 21",   group=grpEMA)
ema9Color   = input.color(color.blue, "EMA 9 Color",           group=grpEMA, inline="ema9")
ema9Width   = input.int(2,            "Width",                 minval=1, maxval=5, group=grpEMA, inline="ema9")
ema21Color  = input.color(color.red,  "EMA 21 Color",          group=grpEMA, inline="ema21")
ema21Width  = input.int(2,            "Width",                 minval=1, maxval=5, group=grpEMA, inline="ema21")

grpFilter = "Entry Filters"
useEmaFilter = input.bool(true, "Require Close > EMA 9 AND EMA 21", group=grpFilter)
useVolFilter = input.bool(true, "Require Volume > 1.2x 20-bar Avg", group=grpFilter,
     tooltip="Auto-bypassed on symbols with no volume data (e.g. NIFTY / BANKNIFTY / SENSEX index charts). Chart the futures instead if you want the volume check active.")

grpStyle = "Display"
extendBars    = input.int(30, "Extend Zones Right (bars)", minval=5, maxval=200, group=grpStyle)
showInfoTable = input.bool(false, "Show Info Table", group=grpStyle)
showABLabels  = input.bool(true,  "Show A / B Labels", group=grpStyle)
showExpired   = input.bool(false, "Show Expired Setups (price already past)", group=grpStyle)
showCompletedLabels = input.bool(false, "Show Completed Trade Labels (backtest)", group=grpStyle,
     tooltip="When ON, a trade that finished (✓ target hit) or was missed (✖ price overran its target zone without an entry) leaves behind just its dim peak label — no zones or lines. Handy for backtesting; keep OFF for a clean live chart.")
showZigZagDbg = input.bool(false, "Show Peak ZigZag (debug)", group=grpStyle)
showTrade2 = input.bool(true, "Show Trade 2", group=grpStyle)
showTrade3 = input.bool(true, "Show Trade 3", group=grpStyle)
showTrade4 = input.bool(true, "Show Trade 4", group=grpStyle)

grpZones = "Zone Styling (unified for all trades)"
colEntry  = input.color(#D4AF37, "Entry Zone Color (golden)", group=grpZones)
colTarget = input.color(#22AB94, "Target Zone Color (green)", group=grpZones)
colSL     = input.color(color.red, "SL Line Color (red)",     group=grpZones)
colB      = input.color(#22AB94, "B Line Color",              group=grpZones)
zoneBorderWidth = input.int(2, "Zone Border Width", minval=1, maxval=5, group=grpZones)
transpActive = input.int(55, "Active Zone Transparency", minval=0,  maxval=100, group=grpZones)
transpFaint  = input.int(88, "Faint Zone Transparency",  minval=50, maxval=100, group=grpZones)
slWidth      = input.int(1, "SL Line Width", minval=1, maxval=5, group=grpZones)
dimGray      = input.color(color.gray, "Faint Outline Color", group=grpZones)

// =============== ZIGZAG ENGINES ===============
// zzB  -> defines the common low B (uses Min Decline %)
// zzP  -> finds the nested peaks (uses the finer Peak Sensitivity %)
var settingsB = zigzag.Settings.new(minDeclinePct, zigDepth, color(na), false, false, false, false, "Absolute", true)
var zigzag.ZigZag zzB = zigzag.newInstance(settingsB)
settingsB.devThreshold := minDeclinePct
settingsB.depth := zigDepth
zzB.update()

var settingsP = zigzag.Settings.new(pivotSensPct, zigDepth, color(na), false, false, false, false, "Absolute", true)
var zigzag.ZigZag zzP = zigzag.newInstance(settingsP)
settingsP.devThreshold := pivotSensPct
settingsP.depth := zigDepth
zzP.update()

// =============== EMAs ===============
ema9  = ta.ema(close, 9)
ema21 = ta.ema(close, 21)
plot(ema9,  "EMA 9",  color = showEMAs ? ema9Color  : color.new(ema9Color,  100), linewidth=ema9Width)
plot(ema21, "EMA 21", color = showEMAs ? ema21Color : color.new(ema21Color, 100), linewidth=ema21Width)

// =============== VOLUME AVAILABILITY (index auto-bypass) ===============
// Index symbols (NIFTY / BANKNIFTY / SENSEX) carry no volume. Without this
// bypass the volume filter could never pass and no entry would ever fire.
var bool symHasVol = false
if not na(volume) and volume > 0
    symHasVol := true

// =============== PERSISTENT STATE ===============
var float B_price = na
var int   B_time  = na

var float A0_price = na
var int   A0_time  = na
var float A1_price = na
var int   A1_time  = na
var float A2_price = na
var int   A2_time  = na
var float A3_price = na
var int   A3_time  = na
var int   activeSwings = 0

var int st0 = 0
var int st1 = 0
var int st2 = 0
var int st3 = 0
var int stateForB = na

// =============== DETECTION: common low B (latest confirmed >=Min Decline%) ===============
if zzB.pivots.size() >= 1
    float fB  = na
    int   fBt = na
    for i = 0 to zzB.pivots.size() - 1
        p = zzB.pivots.get(i)
        if p.end.price < p.start.price  // down-swing -> its end is the low
            fB  := p.end.price
            fBt := p.end.time
    isFirst   = not na(fBt) and na(B_time)
    isChanged = not na(fBt) and not na(B_time) and fBt != B_time
    if isFirst or isChanged
        B_price := fB
        B_time  := fBt

bool hasB = not na(B_price)

// =============== DETECTION: nested peaks (nearest-first, finer engine) ===============
if hasB and zzP.pivots.size() >= 1
    array<float> ptP = array.new<float>()
    array<int>   ptT = array.new<int>()
    array<bool>  ptH = array.new<bool>()
    for i = 0 to zzP.pivots.size() - 1
        p = zzP.pivots.get(i)
        bool down = p.end.price < p.start.price
        if p.start.time == p.end.time
            // V1.1: the library's SEED pivot (start==end) carries no direction —
            // infer it from the segment that leaves this point instead
            bool seedHigh = false
            if i + 1 < zzP.pivots.size()
                pn = zzP.pivots.get(i + 1)
                seedHigh := pn.end.price < pn.start.price
            if array.indexof(ptT, p.end.time) == -1
                array.push(ptT, p.end.time)
                array.push(ptP, p.end.price)
                array.push(ptH, seedHigh)
        else
            if array.indexof(ptT, p.start.time) == -1
                array.push(ptT, p.start.time)
                array.push(ptP, p.start.price)
                array.push(ptH, down)        // down-swing start = HIGH
            if array.indexof(ptT, p.end.time) == -1
                array.push(ptT, p.end.time)
                array.push(ptP, p.end.price)
                array.push(ptH, not down)    // down-swing end = LOW

    float k0 = na
    float k1 = na
    float k2 = na
    float k3 = na
    int   k0t = na
    int   k1t = na
    int   k2t = na
    int   k3t = na
    float lastKept = na
    int   kept = 0
    float gapMul = 1.0 + minGapPct / 100.0
    int   nPts = array.size(ptT)
    for j = 0 to nPts - 1
        idx = nPts - 1 - j
        int t = array.get(ptT, idx)
        if t < B_time
            float pr  = array.get(ptP, idx)
            bool  isH = array.get(ptH, idx)
            if isH
                if pr > B_price * gapMul
                    if na(lastKept)
                        // T1 = nearest peak above B
                        k0  := pr
                        k0t := t
                        lastKept := pr
                        kept := 1
                    else if pr > lastKept * gapMul
                        // clearly higher -> new nested peak
                        if kept == 1
                            k1  := pr
                            k1t := t
                        else if kept == 2
                            k2  := pr
                            k2t := t
                        else if kept == 3
                            k3  := pr
                            k3t := t
                        lastKept := pr
                        kept += 1
                    else if pr > lastKept
                        // higher but within the gap -> keep the HIGHER one
                        if kept == 1
                            k0  := pr
                            k0t := t
                        else if kept == 2
                            k1  := pr
                            k1t := t
                        else if kept == 3
                            k2  := pr
                            k2t := t
                        else if kept == 4
                            k3  := pr
                            k3t := t
                        lastKept := pr
            else
                if pr < B_price
                    break                // deeper low -> different decline, stop
            if kept >= maxSwings
                break

    A0_price := k0
    A0_time  := k0t
    A1_price := k1
    A1_time  := k1t
    A2_price := k2
    A2_time  := k2t
    A3_price := k3
    A3_time  := k3t
    activeSwings := kept

bool has0 = hasB and not na(A0_price)
bool has1 = hasB and not na(A1_price)
bool has2 = hasB and not na(A2_price)
bool has3 = hasB and not na(A3_price)

// =============== STATE RESET on new B ===============
isFirstStateInit = hasB and na(stateForB)
isStateBChanged  = hasB and not na(stateForB) and B_time != stateForB
if isFirstStateInit or isStateBChanged
    stateForB := B_time
    st0 := 1
    st1 := 1
    st2 := 1
    st3 := 1

// =============== PER-SWING FIB LEVELS ===============
swingLevels(float Ap, float Bp) =>
    float r  = Ap - Bp
    float sl = Bp + fib_sl1     * r
    float eL = Bp + fib_entry1L * r
    float eH = Bp + fib_entry1H * r
    float tL = Bp + fib_tp1L    * r
    float tH = Bp + fib_tp1H    * r
    [sl, eL, eH, tL, tH]

[sl0, e0L, e0H, t0L, t0H] = swingLevels(A0_price, B_price)
[sl1, e1L, e1H, t1L, t1H] = swingLevels(A1_price, B_price)
[sl2, e2L, e2H, t2L, t2H] = swingLevels(A2_price, B_price)
[sl3, e3L, e3H, t3L, t3H] = swingLevels(A3_price, B_price)

// =============== FILTERS ===============
volAvg = ta.sma(volume, 20)
// volume filter auto-bypasses on symbols without volume data (indices)
volOK  = not useVolFilter or not symHasVol or volume > volAvg * 1.2
emaOK  = not useEmaFilter or (close > ema9 and close > ema21)

// =============== TRADE STATE MACHINE (single trade per swing) ===============
stepTrade(int s, float eHi, float tLo, float slLvl, bool eOK, bool vOK) =>
    int  ns = s
    bool eFire = false
    bool tFire = false
    bool sFire = false
    bool crossE = not na(eHi) and close > eHi and close[1] <= eHi
    if s == 1 and crossE and eOK and vOK
        ns := 2
        eFire := true
    else if s == 2
        if not na(tLo) and high >= tLo
            ns := 3
            tFire := true
        else if not na(slLvl) and close < slLvl
            ns := 1
            sFire := true
    [ns, eFire, tFire, sFire]

bool e0 = false
bool p0 = false
bool x0 = false
bool e1 = false
bool p1 = false
bool x1 = false
bool e2 = false
bool p2 = false
bool x2 = false
bool e3 = false
bool p3 = false
bool x3 = false

if hasB and barstate.isconfirmed
    if has0
        [ns0, ef0, tf0, sf0] = stepTrade(st0, e0H, t0L, sl0, emaOK, volOK)
        st0 := ns0
        e0  := ef0
        p0  := tf0
        x0  := sf0
    if has1
        [ns1, ef1, tf1, sf1] = stepTrade(st1, e1H, t1L, sl1, emaOK, volOK)
        st1 := ns1
        e1  := ef1
        p1  := tf1
        x1  := sf1
    if has2
        [ns2, ef2, tf2, sf2] = stepTrade(st2, e2H, t2L, sl2, emaOK, volOK)
        st2 := ns2
        e2  := ef2
        p2  := tf2
        x2  := sf2
    if has3
        [ns3, ef3, tf3, sf3] = stepTrade(st3, e3H, t3L, sl3, emaOK, volOK)
        st3 := ns3
        e3  := ef3
        p3  := tf3
        x3  := sf3
    // V1.1: RETIRE MISSED TRADES — a WAITING trade whose whole target zone price has
    // closed above can never legitimately enter; mark 4 (✖ missed, permanent) so the
    // spotlight moves to the next trade.
    if has0 and st0 == 1 and close > t0H
        st0 := 4
    if has1 and st1 == 1 and close > t1H
        st1 := 4
    if has2 and st2 == 1 and close > t2H
        st2 := 4
    if has3 and st3 == 1 and close > t3H
        st3 := 4

// active trade = lowest swing still tradeable (not TP'd, not missed);
// expiry = price above highest peak
int activeIdx = na
if has0 and st0 < 3
    activeIdx := 0
else if has1 and st1 < 3
    activeIdx := 1
else if has2 and st2 < 3
    activeIdx := 2
else if has3 and st3 < 3
    activeIdx := 3

float topA = has3 ? A3_price : has2 ? A2_price : has1 ? A1_price : has0 ? A0_price : na
bool setupExpired = hasB and not na(topA) and close > topA

// =============== PLOT SHAPES ===============
plotshape(e0 or e1 or e2 or e3, "Entry", style=shape.triangleup,   location=location.belowbar, color=color.new(colEntry, 0),  size=size.small, text="▲ Entry", textcolor=color.white)
plotshape(p0 or p1 or p2 or p3, "TP",    style=shape.triangledown, location=location.abovebar, color=color.new(colTarget, 0), size=size.small, text="TP",     textcolor=color.white)
plotshape(x0 or x1 or x2 or x3, "SL",    style=shape.xcross,       location=location.abovebar, color=color.new(colSL, 0),     size=size.small, text="SL",     textcolor=color.white)

// =============== DRAWING ===============
var array<box>   gBoxes  = array.new<box>()
var array<line>  gLines  = array.new<line>()
var array<label> gLabels = array.new<label>()

clearDrawings() =>
    for b in gBoxes
        box.delete(b)
    for l in gLines
        line.delete(l)
    for lb in gLabels
        label.delete(lb)
    array.clear(gBoxes)
    array.clear(gLines)
    array.clear(gLabels)

drawSwing(float Ap, int At, float eL, float eH, float tL, float tH, float slv, bool isActive, bool doShow, int rightT, string tag) =>
    if not na(Ap) and doShow and not na(B_time)
        int transp = isActive ? transpActive : transpFaint
        color tBorder = isActive ? colTarget : color.new(dimGray, 75)
        // target (green) zone for both active and faint
        array.push(gBoxes, box.new(B_time, tH, rightT, tL, xloc=xloc.bar_time, border_color=tBorder, bgcolor=color.new(colTarget, transp), border_width=isActive ? zoneBorderWidth : 1))
        // peak (A) line
        array.push(gLines, line.new(At, Ap, rightT, Ap, xloc=xloc.bar_time, color=color.new(isActive ? colTarget : dimGray, isActive ? 25 : 70), style=line.style_dotted, width=1))
        if showABLabels
            array.push(gLabels, label.new(At, Ap, tag + "  " + str.tostring(Ap, format.mintick), xloc=xloc.bar_time, style=label.style_label_down, color=color.new(isActive ? colTarget : dimGray, 15), textcolor=color.white, size=size.small))
        if isActive
            array.push(gBoxes, box.new(B_time, eH, rightT, eL, xloc=xloc.bar_time, border_color=colEntry, bgcolor=color.new(colEntry, transpActive), border_width=zoneBorderWidth))
            array.push(gLines, line.new(B_time, slv, rightT, slv, xloc=xloc.bar_time, color=color.new(colSL, 30), style=line.style_dashed, width=slWidth))
            int midT = int(B_time + (rightT - B_time) / 2)
            array.push(gLabels, label.new(midT, (eL + eH) / 2, tag + " Entry", xloc=xloc.bar_time, style=label.style_none, textcolor=colEntry, size=size.small))
            array.push(gLabels, label.new(midT, (tL + tH) / 2, tag + " Target", xloc=xloc.bar_time, style=label.style_none, textcolor=colTarget, size=size.small))

// finished trade -> just a dim peak label: "✓" target hit, "✖" missed (V1.1)
drawDoneLabel(float Ap, int At, string tag, bool hitTP) =>
    if not na(Ap)
        array.push(gLabels, label.new(At, Ap, (hitTP ? "✓ " : "✖ ") + tag + "  " + str.tostring(Ap, format.mintick), xloc=xloc.bar_time, style=label.style_label_down, color=color.new(dimGray, 25), textcolor=color.white, size=size.small))

if barstate.islast
    clearDrawings()
    if hasB and (showExpired or not setupExpired)
        int rightTime = int(time + extendBars * (time - time[1]))
        array.push(gLines, line.new(B_time, B_price, rightTime, B_price, xloc=xloc.bar_time, color=color.new(colB, 20), style=line.style_dotted, width=1))
        if showABLabels
            array.push(gLabels, label.new(B_time, B_price, "B  " + str.tostring(B_price, format.mintick), xloc=xloc.bar_time, style=label.style_label_up, color=color.new(colB, 15), textcolor=color.white, size=size.small))
        if has0
            if st0 < 3
                drawSwing(A0_price, A0_time, e0L, e0H, t0L, t0H, sl0, activeIdx == 0, true,       rightTime, "T1")
            else if showCompletedLabels
                drawDoneLabel(A0_price, A0_time, "T1", st0 == 3)
        if has1
            if st1 < 3
                drawSwing(A1_price, A1_time, e1L, e1H, t1L, t1H, sl1, activeIdx == 1, showTrade2, rightTime, "T2")
            else if showCompletedLabels and showTrade2
                drawDoneLabel(A1_price, A1_time, "T2", st1 == 3)
        if has2
            if st2 < 3
                drawSwing(A2_price, A2_time, e2L, e2H, t2L, t2H, sl2, activeIdx == 2, showTrade3, rightTime, "T3")
            else if showCompletedLabels and showTrade3
                drawDoneLabel(A2_price, A2_time, "T3", st2 == 3)
        if has3
            if st3 < 3
                drawSwing(A3_price, A3_time, e3L, e3H, t3L, t3H, sl3, activeIdx == 3, showTrade4, rightTime, "T4")
            else if showCompletedLabels and showTrade4
                drawDoneLabel(A3_price, A3_time, "T4", st3 == 3)
    if showZigZagDbg and zzP.pivots.size() >= 1
        for i = 0 to zzP.pivots.size() - 1
            p = zzP.pivots.get(i)
            array.push(gLines, line.new(p.start.time, p.start.price, p.end.time, p.end.price, xloc=xloc.bar_time, color=color.new(color.yellow, 50), width=1, style=line.style_dashed))

// =============== INFO TABLE ===============
var table info = table.new(position.top_right, 5, 16, bgcolor=color.new(color.black, 70), border_width=1)

stateLabel(int s) =>
    s == 1 ? "⏳ wait" : s == 2 ? "🟢 IN" : s == 3 ? "✓ TP" : s == 4 ? "✖ missed" : "—"

stateColor(int s) =>
    s == 2 ? color.lime : s == 1 ? color.aqua : s == 3 ? color.gray : s == 4 ? color.orange : color.white

fillCol(int col, bool has, float Ap, float eL, float eH, float tL, float tH, float slv, int st) =>
    float dep = has and Ap > 0 ? (Ap - B_price) / Ap * 100.0 : na
    table.cell(info, col, 1, has ? str.tostring(Ap, format.mintick) : "—", text_color=color.white, text_size=size.small)
    table.cell(info, col, 2, has ? str.tostring(eL, format.mintick) + "–" + str.tostring(eH, format.mintick) : "—", text_color=colEntry, text_size=size.small)
    table.cell(info, col, 3, has ? str.tostring(tL, format.mintick) + "–" + str.tostring(tH, format.mintick) : "—", text_color=colTarget, text_size=size.small)
    table.cell(info, col, 4, has ? str.tostring(slv, format.mintick) : "—", text_color=colSL, text_size=size.small)
    table.cell(info, col, 5, has ? str.tostring(dep, "#.##") + "%" : "—", text_color=color.gray, text_size=size.small)
    table.cell(info, col, 6, has ? stateLabel(st) : "—", text_color=stateColor(st), text_size=size.small)
